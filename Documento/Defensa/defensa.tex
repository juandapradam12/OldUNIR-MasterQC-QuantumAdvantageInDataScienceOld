\section{Introducción}
% =====================================================================
\begin{frame}{Introducción --- planteamiento del problema}
El problema central del trabajo \citep{preskill2018, biamonte2017}:

\begin{quote}
\itshape
\textquestiondown{}En qué condiciones, y para qué tipos de problemas de ciencia de datos,
los algoritmos cuánticos disponibles en la era NISQ ofrecen una
ventaja real y medible frente a los algoritmos clásicos?
\end{quote}

\vspace{0.6em}
\begin{block}{Hipótesis operativa}
La ventaja es \textbf{real pero condicionada}:
\begin{enumerate}
  \item coincidencia \textbf{feature map} $\leftrightarrow$ \textbf{generador de datos}, o
  \item problema formulable como QUBO con arquitectura \textbf{híbrida} a escala viable.
\end{enumerate}
\end{block}
\end{frame}

\begin{frame}{Introducción --- alcance}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Familias industriales (4)}
\begin{itemize}
  \item Clasificación supervisada (QSVM)
  \item Regresión (QK-SVR)
  \item Optimización combinatoria (QAOA)
  \item Optimización / híbrido (VQE + MDS)
\end{itemize}
{\footnotesize Extendidas a 9 protocolos justos en el documento (anomalías, TSP, lags,\ldots).}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Tres capas de datos}
\begin{enumerate}
  \item Sintéticos con estructura cuántica nativa \citep{liu2021}
  \item Práctica SKG Tecnología
  \item Referencia pública
\end{enumerate}
\end{column}
\end{columns}
\end{frame}

\section{Contexto y estado de la técnica}
% =====================================================================
\begin{frame}{Era NISQ}
\begin{block}{NISQ \citep{preskill2018}}
Dispositivos de escala intermedia \textbf{ruidosos}: qubits limitados, decoherencia, sin corrección de errores a gran escala.
\end{block}
\begin{itemize}
  \item Algoritmos viables hoy: \textbf{variacionales} (QAOA, VQE) e híbridos \citep{farhi2014, cerezo2021}.
  \item Kernels cuánticos: acceso a espacios de Hilbert de alta dimensión \citep{schuld2021, huang2021}.
  \item Distinción clave: \textbf{metodologías} maduras vs.\ \textbf{hardware} aún inmaduro.
\end{itemize}
\end{frame}

\begin{frame}{Comparaciones cuántico--clásico en la literatura}
\begin{itemize}
  \item Condiciones teóricas de ventaja de kernel: geometría de datos crítica \citep{huang2021}.
  \item Datasets donde el kernel cuántico es óptimo por construcción \citep{liu2021}.
  \item Concentración de kernels y límites prácticos \citep{thanasilp2022}.
  \item Gap empírico: pocos estudios \textbf{justos} sobre datos empresariales reales \citep{fastovets2019}.
\end{itemize}
\begin{alertblock}{Posicionamiento de este TFM}
Evidencia empírica + protocolos justos + diagnóstico $\mathcal{Q}(\mathcal{D})$ + QCR formalmente planteado.
\end{alertblock}
\end{frame}

\centering
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{Familia} & \textbf{Cuántico} & \textbf{Clásico de referencia} \\
\midrule
Clasificación & QSVM (ZZ) & GB, RF, SVM-RBF \\
Regresión & QK-SVR & XGBoost, RF, SVR-RBF, Ridge \\
Optimización (asignación) & QAOA ($p{=}1$) & Greedy, búsqueda local, ILP \\
Portafolio / cobertura & VQE / QAOA-MDS & Markowitz, fuerza bruta \\
Anomalías / lags & OCSVM-ZZ / QK-SVR & IF, LOF, Ridge, GB \\
\bottomrule
\end{tabular}

\vspace{0.8em}
Cada familia se evalúa, cuando aplica, en:
\quad (i) \textbf{cuántico-nativo}, \quad (ii) \textbf{SKG}, \quad (iii) \textbf{referencia pública}.

\vspace{0.5em}
\begin{block}{Condición operativa (sin detenerse en formalismo)}
Kernels: ventaja si el \textbf{generador} coincide con el feature map (ZZ medido aquí).
Optimización: QUBO + loop híbrido (QAOA/VQE + COBYLA) a escala viable.
\end{block}
\end{frame}

\section{Conjeturas}
% =====================================================================
\begin{frame}{De la evidencia emergen dos preguntas}
Tras los 9 casos \textbf{no} partimos de $\mathcal{Q}$ ni de QCR: partimos de hechos.

\begin{enumerate}
  \item Ventaja de kernel \textbf{clara} solo cuando el generador coincide con el feature map (ZZ-nativo; también \texttt{zz\_phase}, \texttt{zz\_lag}).
  \item En SKG/Telco (sin esa estructura) gana GB/RF: \textbf{no} son cuántico-nativos.
  \item En QUBO híbrido hay valor a escala moderada; TSP $n{=}4$ empata al óptimo (modelado $\neq$ speedup).
\end{enumerate}

\vspace{0.4em}
\begin{block}{Preguntas originales (Cap.~6)}
\begin{enumerate}
  \item \textbf{P1 --- tests de cuanticidad:}
  \textquestiondown{}Se puede diagnosticar \emph{a priori} si un dataset admitirá ventaja?
  \item \textbf{P2 --- mapeos cuántico-compatibles:}
  Si no hay estructura nativa, \textquestiondown{}existe un $\psi$ que la induzca sin destruir la señal?
\end{enumerate}
\end{block}
\end{frame}

\begin{frame}{Camino a la Conjetura 1 (antes de la fórmula)}
\begin{enumerate}
  \item \textbf{Literatura:} un solo indicador no basta \citep{huang2021, liu2021, thanasilp2022}.
  \item \textbf{Hecho empírico:} en nativo hay ventaja; en SKG/Telco no (vs.\ GB/RF).
  \item \textbf{Notebook~08:} medimos $\Delta A$, $d_F$, $H(K)$ sobre los tres datasets.
  \item \textbf{Hallazgo:} $d_F$ rankea mejor; $\Delta A$ se degrada con PCA a 2 qubits.
  \item \textbf{Barrido sintético} en $\alpha$ (fracción de estructura ZZ):
        $\rho_S(\mathcal{Q},\mathrm{ventaja})=0.833$ ($p{=}0.005$, $n{=}9$).
  \item \textbf{Propuesta:} un score \emph{compuesto} e interpretable --- no un oráculo de despliegue.
\end{enumerate}
\end{frame}

\begin{frame}{Camino a la Conjetura 2 / QCR (antes de la fórmula)}
\begin{enumerate}
  \item Si el diagnóstico dice ``clásico'', el default es el stack clásico.
  \item Pero la pregunta correcta es:
        \textquestiondown{}falla el \textbf{dato} o el \textbf{espacio de representación}?
  \item Hipótesis: existe $\psi:\mathcal{X}\to\mathcal{Z}$ polinómica que induzca
        geometría usable por el feature map \textbf{sin} destruir la señal.
  \item \textbf{Notebook~09:} cuatro prototipos --- ZZ-compatibles, rotación $\max d_F$,
        re-uploading, Q-UMAP/MDS.
  \item Resultado: subidas locales de $\mathcal{Q}$ \emph{vs.\ RBF}; bajo $\mathcal{Q}_{\mathrm{rob}}$
        Telco sigue \texttt{classical} y SKG solo cambia geometría (\texttt{try-ZZ}),
        \textbf{sin} lift de accuracy vs.\ GB/RF.
\end{enumerate}
\begin{alertblock}{Lectura}
QCR es un \textbf{programa de investigación} formalizado desde la evidencia --- no un algoritmo cerrado ni un rescate operativo.
\end{alertblock}
\end{frame}

\begin{frame}{Conjetura 2 --- espacio de representación QCR}
Si $\mathcal{Q}(\mathcal{D})\le 0$, se conjetura $\mathcal{Z}$ y $\psi:\mathcal{X}\to\mathcal{Z}$ \textbf{polinómica} tales que
\begin{equation}
  \mathbf{x}\xrightarrow{\psi}\mathbf{z}\xrightarrow{f_Q^\star}\hat{y}
  \qquad\text{con}\quad
  \mathcal{Q}(\psi(\mathcal{D}))>0
  \;\;\text{y}\;\;
  \varepsilon(f_Q^\star\circ\psi)\le\varepsilon(f_C^\star).
\end{equation}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Intuición}
Transformar el dato \emph{antes} del kernel para que el mapa ``vea'' estructura usable.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{alertblock}{Estado}
Conjetura \textbf{abierta}.\\
Este TFM la \textbf{plantea} + evidencia preliminar --- no demostración.
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

\section{Conclusiones}
% =====================================================================
\begin{frame}{Conclusiones (I) --- ventaja condicionada}
\begin{enumerate}
  \item La ventaja cuántica en ciencia de datos es \textbf{real pero condicionada}.
  \item En kernels: aparece cuando el feature map coincide con el generador
        (ZZ medido aquí; Pauli-Y confirma el mismo principio).
  \item En \textbf{datos reales} SKG/Telco \textbf{sin} esa estructura: gana GB/RF
        (no son cuántico-nativos).
  \item Regresión: ventaja acotada (Friedman); en movilidad SKG, árboles dominan.
  \item Anomalías / lags: misma regla --- solo con target alineado
        (\texttt{zz\_phase}, \texttt{zz\_lag}).
\end{enumerate}
\end{frame}

\begin{frame}{Conclusiones (II) --- híbridos y coste}
\begin{enumerate}
  \setcounter{enumi}{5}
  \item QAOA reduce coste de asignación $\approx 30\%$ vs.\ Greedy a escala moderada.
  \item VQE competitivo en portafolio pequeño; TSP $n{=}4$: empate óptimo
        (valor de \emph{modelado}, no speedup).
  \item MDS: patrón híbrido viable ($P(\mathrm{opt})\approx 73\%$).
  \item Coste de simulación del kernel $O(n^2)$: obstáculo práctico actual.
  \item Metodologías maduras vs.\ hardware inmaduro: resultados mixtos son esperables en NISQ.
\end{enumerate}
\end{frame}

\begin{frame}{Conclusiones (III) --- conjeturas y adopción}
\begin{enumerate}
  \setcounter{enumi}{10}
  \item $\mathcal{Q}(\mathcal{D})$: diagnóstico útil en barrido sintético ($\rho_S{=}0.833$);
        en reales, $n$ pequeño y matizado por $\mathcal{Q}_{\mathrm{rob}}$.
  \item QCR: \textbf{planteado formalmente}; evidencia preliminar; no demostración ni rescate operativo.
  \item Adopción recomendada: \textbf{selectiva e híbrida} --- módulo cuántico como
        herramienta adicional, no sustituto del stack clásico (SKG).
\end{enumerate}
\vspace{0.6em}
\begin{block}{Respuesta a la pregunta del TFM}
Ventaja medible hoy cuando hay coincidencia mapa--generador o QUBO híbrido viable;
en ausencia de eso, el clásico domina --- y eso también es un resultado.
\end{block}
\end{frame}

\begin{frame}{Limitaciones (I)}
\begin{itemize}
  \item \textbf{Simulación statevector}: sin ruido NISQ de hardware; ideales $\neq$ shots reales.
  \item \textbf{PCA a $n_q{=}2$}: $\sim$68\% de varianza descartada; $\mathcal{Q}$ mide el subespacio proyectado.
  \item \textbf{Muestra} $n{\approx}80$ en indicadores; correlación en 3 datasets reales es orientativa.
  \item \textbf{Baseline RBF}: $\mathcal{Q}>0$ vs.\ RBF $\neq$ ventaja vs.\ GB/RF (caso SKG).
\end{itemize}
\end{frame}

\begin{frame}{Limitaciones (II)}
\begin{itemize}
  \item \textbf{QCR} sin demostración fuerte; Q-UMAP sin métrica cuántica real (MDS proxy).
  \item \textbf{Un feature map operativo principal} (ZZ), con validación cruzada Pauli-Y limitada.
  \item \textbf{TSP}: empate óptimo $\neq$ speedup asintótico.
  \item \textbf{Escalabilidad}: kernel $O(n^2)$ limita $n$ en simulación; no se evaluó hardware real.
  \item \textbf{Datos SKG}: fraction de la operación; no cierran todos los casos de uso empresariales.
\end{itemize}
\end{frame}
